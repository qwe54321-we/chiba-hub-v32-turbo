-- เพิ่มเติม: Whitelist + License Key + Expiration Date + Hardware Check
-- ====================================================================

local CHIBA_LICENSES = {
    -- [USERNAME] = { key = "LICENSE_KEY", expiry = "YYYY-MM-DD", status = "ACTIVE" }
    ["qwe54321-we"] = {
        key = "CHIBA_LICENSE_OWNER_001_PREMIUM",
        expiry = "2030-12-31",
        status = "ACTIVE",
        tier = "OWNER"
    },
    ["Tester1"] = {
        key = "CHIBA_LICENSE_TESTER_001",
        expiry = "2026-12-31",
        status = "ACTIVE",
        tier = "TESTER"
    },
    ["Tester2"] = {
        key = "CHIBA_LICENSE_TESTER_002",
        expiry = "2026-09-30",
        status = "ACTIVE",
        tier = "TESTER"
    },
}

-- ====================================================================
-- 🔍 VERIFICATION FUNCTIONS
-- ====================================================================

local function getCurrentDate()
    return os.date("%Y-%m-%d")
end

local function isLicenseExpired(expiryDate)
    local currentDate = getCurrentDate()
    return currentDate > expiryDate
end

local function getHardwareID()
    -- สร้าง Hardware ID จากตัวรัน (ถ้าเป็น Executor ที่รองรับ)
    local hwid = ""
    pcall(function()
        hwid = tostring(game:GetService("UserInputService")):sub(1, 16)
    end)
    return hwid ~= "" and hwid or "EXECUTOR_UNKNOWN"
end

local function verifyLicense(username, licenseKey)
    if not CHIBA_LICENSES[username] then
        return false, "❌ Username not found in whitelist"
    end

    local license = CHIBA_LICENSES[username]

    -- ตรวจสอบ Key
    if license.key ~= licenseKey then
        return false, "❌ Invalid license key"
    end

    -- ตรวจสอบ Status
    if license.status ~= "ACTIVE" then
        return false, "❌ License is " .. license.status
    end

    -- ตรวจสอบวันหมดอายุ
    if isLicenseExpired(license.expiry) then
        return false, "❌ License expired on " .. license.expiry .. ". Please renew."
    end

    return true, "✅ License verified successfully (Tier: " .. license.tier .. ")"
end

local function authenticateUser()
    local Players = game:GetService("Players")
    local lp = Players.LocalPlayer
    local username = lp.Name

    print("[CHIBA AUTH] Verifying user: " .. username)
    print("[CHIBA AUTH] Current date: " .. getCurrentDate())
    print("[CHIBA AUTH] Hardware ID: " .. getHardwareID())

    if not CHIBA_LICENSES[username] then
        warn("[CHIBA AUTH] ❌ User not authorized!")
        return false
    end

    local license = CHIBA_LICENSES[username]
    local isValid, message = verifyLicense(username, license.key)

    print("[CHIBA AUTH] " .. message)

    if not isValid then
        warn("[CHIBA AUTH] Access Denied!")
        return false
    end

    print("[CHIBA AUTH] ✅ Access Granted!")
    return true
end

-- ====================================================================
-- 🛡️ LICENSE MANAGEMENT FUNCTIONS (Admin Only)
-- ====================================================================

local function addLicense(username, licenseKey, expiryDate, tier)
    if not CHIBA_LICENSES["qwe54321-we"] then
        warn("Admin function only!")
        return false
    end

    CHIBA_LICENSES[username] = {
        key = licenseKey,
        expiry = expiryDate,
        status = "ACTIVE",
        tier = tier or "USER"
    }

    print("✅ License added for: " .. username)
    return true
end

local function removeLicense(username)
    if CHIBA_LICENSES[username] then
        CHIBA_LICENSES[username] = nil
        print("✅ License removed for: " .. username)
        return true
    end
    return false
end

local function suspendLicense(username)
    if CHIBA_LICENSES[username] then
        CHIBA_LICENSES[username].status = "SUSPENDED"
        print("⚠️ License suspended for: " .. username)
        return true
    end
    return false
end

local function renewLicense(username, newExpiryDate)
    if CHIBA_LICENSES[username] then
        CHIBA_LICENSES[username].expiry = newExpiryDate
        CHIBA_LICENSES[username].status = "ACTIVE"
        print("✅ License renewed for: " .. username .. " until " .. newExpiryDate)
        return true
    end
    return false
end

local function listAllLicenses()
    print("\n=== 📋 CHIBA LICENSE DATABASE ===")
    for username, data in pairs(CHIBA_LICENSES) do
        local status = data.status
        local icon = status == "ACTIVE" and "✅" or "❌"
        local expired = isLicenseExpired(data.expiry) and " [EXPIRED]" or ""
        print(icon .. " " .. username .. " | Tier: " .. data.tier .. " | Expiry: " .. data.expiry .. expired)
    end
    print("================================\n")
end

-- ====================================================================
-- 🚀 AUTO-AUTH ON SCRIPT LOAD
-- ====================================================================

local function bootScript()
    print("\n" .. string.rep("=", 50))
    print("🔐 CHIBA HUB GOD - LICENSE AUTHENTICATION")
    print(string.rep("=", 50))

    if authenticateUser() then
        print("🚀 Loading main script...\n")
        return true
    else
        print("🚫 Script execution blocked - Invalid license\n")
        print(string.rep("=", 50) .. "\n")
        return false
    end
end

_G.ChibaAuth = {

    authenticate = authenticateUser,
    addLicense = addLicense,
    removeLicense = removeLicense,
    suspendLicense = suspendLicense,
    renewLicense = renewLicense,
    listLicenses = listAllLicenses,
    boot = bootScript,
    getHWID = getHardwareID,
}

-- เช็คให้ครบทั้ง 3 ฟังก์ชันในที่เดียว แล้วใช้ตัวแปรนี้ทุกจุด (แก้ปัญหาเดิมที่ save เช็คแค่ writefile

local hasFileIO = type(writefile) == "function" and type(readfile) == "function" and type(isfile) == "function"

-- โหมดเก็บในเครื่องผ่านตัวรัน (Executor Workspace Mode)

local points = {}
local nextId = 1
local SAVE_FILENAME = "CheckpointData_" .. player.UserId .. ".json" -- แยกไฟล์ตาม UID ป้องกันการตีกันถ้าสลับไอดี
local ATTR_NAME = "CheckpointData" -- ไว้สำรองเผื่อตัวรันไม่รองรับ (ชั่วคราวเท่านั้น หายเมื่อออกเกม)

local function loadLocalData()
	local raw = ""

	if hasFileIO then
		if isfile(SAVE_FILENAME) then
			local success, result = pcall(function()
				return readfile(SAVE_FILENAME)
			end)
			if success then
				raw = result
			end
		end
	else
		-- ถ้าไม่ใช่ตัวรัน (หรือรองรับไม่ครบ) ให้กลับไปใช้ Attribute แบบเดิมแก้ขัด
		-- หมายเหตุ: โหมดนี้ไม่ persist ข้ามการออก-เข้าเกม
		raw = player:GetAttribute(ATTR_NAME)
	end

	-- นำข้อความ JSON มาแปลงกลับเป็นตาราง
	if raw and raw ~= "" then
		local success, decoded = pcall(function()
			return HttpService:JSONDecode(raw)
		end)
		if success and decoded then
			points = decoded.points or {}
			nextId = decoded.nextId or 1
		end
	end
end

local function saveLocalData()
	local encoded = HttpService:JSONEncode({points = points, nextId = nextId})

	if hasFileIO then
		pcall(function()
			writefile(SAVE_FILENAME, encoded)
		end)
	else
		player:SetAttribute(ATTR_NAME, encoded)
	end
end

local function getHRP()
	local character = player.Character
	return character and character:FindFirstChild("HumanoidRootPart")
end

local function cframeToTable(cf)
	local rx, ry, rz = cf:ToOrientation()
	return {x = cf.Position.X, y = cf.Position.Y, z = cf.Position.Z, rx = rx, ry = ry, rz = rz}
end

local function tableToCFrame(t)
	return CFrame.new(t.x, t.y, t.z) * CFrame.Angles(t.rx or 0, t.ry or 0, t.rz or 0)
end

local onSyncCallback

local function notify(pointsList, message)
	if onSyncCallback then
		onSyncCallback(pointsList, message)
	end
end

local function doCreate(name)
	if usingServer then
		actionEvent:FireServer("create", {name = name})
		return
	end
	if #points >= MAX_POINTS then
		notify(points, "บันทึกเต็มแล้ว! (สูงสุด " .. MAX_POINTS .. " จุด)")
		return
	end
	local hrp = getHRP()
	if not hrp then
		notify(points, "ตัวละครยังไม่พร้อม ลองใหม่อีกครั้ง")
		return
	end
	local finalName = (name ~= "" and name) or ("จุดที่ " .. tostring(nextId))
	local point = cframeToTable(hrp.CFrame)
	point.id = nextId
	point.name = finalName
	nextId += 1
	table.insert(points, point)

	saveLocalData() -- เซฟข้อมูลลงเครื่องทันที
	notify(points, "บันทึกจุด \"" .. finalName .. "\" แล้ว (" .. #points .. "/" .. MAX_POINTS .. ")")
end

local function doTeleport(id)
	if usingServer then
		actionEvent:FireServer("teleport", {id = id})
		return
	end
	local hrp = getHRP()
	if not hrp then return end
	for _, point in ipairs(points) do
		if point.id == id then
			hrp.CFrame = tableToCFrame(point)
			notify(points, "วาปไปจุด \"" .. point.name .. "\" แล้ว")
			break
		end
	end
end

local function doDelete(id)
	if usingServer then
		actionEvent:FireServer("delete", {id = id})
		return
	end
	for i, point in ipairs(points) do
		if point.id == id then
			table.remove(points, i)
			saveLocalData() -- อัพเดทไฟล์
			notify(points, "ลบจุดแล้ว")
			break
		end
	end
end

local function doRename(id, newName)
	if usingServer then
		actionEvent:FireServer("rename", {id = id, newName = newName})
		return
	end
	for _, point in ipairs(points) do
		if point.id == id and newName ~= "" then
			point.name = newName
			saveLocalData() -- อัพเดทไฟล์
			notify(points, "เปลี่ยนชื่อแล้ว")
			break
		end
	end
end

local function doUpdate(id)
	if usingServer then
		actionEvent:FireServer("update", {id = id})
		return
	end
	local hrp = getHRP()
	if not hrp then
		notify(points, "ตัวละครยังไม่พร้อม ลองใหม่อีกครั้ง")
		return
	end
	for _, point in ipairs(points) do
		if point.id == id then
			local newCF = cframeToTable(hrp.CFrame)
			point.x, point.y, point.z = newCF.x, newCF.y, newCF.z
			point.rx, point.ry, point.rz = newCF.rx, newCF.ry, newCF.rz
			saveLocalData() -- อัพเดทไฟล์
			notify(points, "อัพเดตพิกัดแล้ว")
			break
		end
	end
end

local function doRequestList()
	if usingServer then
		actionEvent:FireServer("requestList", {})
		return
	end
	notify(points, nil)
end

if usingServer then
	syncEvent.OnClientEvent:Connect(function(newPoints, message)
		points = newPoints or {}
		notify(points, message)
	end)
else
	loadLocalData() -- โหลดข้อมูลจากไฟล์ตอนรันสคริปต์
	if hasFileIO then
		print("[CheckpointUI] ใช้โหมดเซฟลงเครื่อง (Workspace ของ Executor) - ข้อมูลจะอยู่ถาวรแม้ออกเกม")
	else
		warn("[CheckpointUI] ตัวรันไม่รองรับ writefile/readfile/isfile ครบทั้งสามฟังก์ชัน -> ใช้โหมดเก็บข้อมูลชั่วคราว (หายเมื่อออกเกม)")
	end
end

local dragging = false
local dragInput = nil
local dragStart, startPos
local dragDistance = 0
local DRAG_THRESHOLD = 8

local function updateIconDrag(input)
	local delta = input.Position - dragStart
	dragDistance = math.max(dragDistance, delta.Magnitude)
	local viewport = workspace.CurrentCamera.ViewportSize
	local newX = math.clamp(startPos.X.Offset + delta.X, 0, viewport.X - icon.AbsoluteSize.X)
	local newY = math.clamp(startPos.Y.Offset + delta.Y, 0, viewport.Y - icon.AbsoluteSize.Y)
	icon.Position = UDim2.new(0, newX, 0, newY)
end

-- แก้บั๊ก multi-touch: เดิมเช็คแค่ UserInputType ทำให้นิ้วที่สอง (หรือการแตะที่อื่นบนจอ)
-- ไปสั่ง dragging=false ก่อนนิ้วแรกปล่อยจริง และ InputChanged เดิมรับ "ทุก" การขยับของ Touch
-- ทั่วจอ (ไม่ใช่แค่ที่กำลังลากไอคอน) ทำให้ไอคอนกระตุกเวลามีสองนิ้วบนจอ
-- ตอนนี้ผูกกับ input object ตัวเดียวที่เริ่มลากจริง (dragInput) ตลอดทั้ง gesture
icon.InputBegan:Connect(function(input)
	if dragging then return end -- กำลังลากอยู่แล้ว ไม่รับนิ้ว/ปุ่มใหม่ซ้อน
	if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = true
		dragStart = input.Position
		startPos = icon.Position
		dragDistance = 0
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging then return end
	-- Touch ใช้ input object เดิมตลอด gesture เทียบ reference ได้ตรงๆ
	-- Mouse ใช้ MouseMovement event แยกต่างหาก (มีเมาส์แค่ตัวเดียวจึงปลอดภัย ไม่ต้องเทียบ reference)
	if input == dragInput or input.UserInputType == Enum.UserInputType.MouseMovement then
		updateIconDrag(input)
	end
end)

icon.InputEnded:Connect(function(input)
	if not dragging then return end
	if input == dragInput then
		dragging = false
		if dragDistance < DRAG_THRESHOLD then
			toggleMenu()
		end
	end
end)

local function selectPoint(id)
	selectedId = id
	homePanel.Visible = false
	pointPanel.Visible = true
	renameConfirming = false
	renameBox.Visible = false
	renameBtn.Text = "เปลี่ยนชื่อ"
	homeBtn.BackgroundColor3 = Color3.fromRGB(45,45,52)

	for _, p in ipairs(currentPoints) do
		if p.id == id then
			pointInfoLabel.Text = p.name .. "\n(" .. math.floor(p.x) .. ", " .. math.floor(p.y) .. ", " .. math.floor(p.z) .. ")"
			break
		end
	end
	for pid, b in pairs(pointButtons) do
		b.BackgroundColor3 = (pid == id) and Color3.fromRGB(50,90,180) or Color3.fromRGB(45,45,52)
	end
end

		btn.LayoutOrder = i

		btn.MouseButton1Click:Connect(function() selectPoint(p.id) end)
		pointButtons[p.id] = btn

	if selectedId then
		local found = false
		for _, p in ipairs(currentPoints) do if p.id == selectedId then found = true break end end
		if found then selectPoint(selectedId) else selectHome() end
	end
end

onSyncCallback = function(newPoints, message)
	currentPoints = newPoints or {}
	rebuildList()
	if message then
		print("[Checkpoint] " .. message)
	end
end

toggleMenu = function()
	menuFrame.Visible = not menuFrame.Visible
	if menuFrame.Visible then
		doRequestList()
	end
end

selectHome()
doRequestList()

local lastUpdate = 0
RunService.Heartbeat:Connect(function()
	if not menuFrame.Visible or selectedId ~= nil then return end
	local now = os.clock()
	if now - lastUpdate < 0.15 then return end
	lastUpdate = now
	local character = player.Character
	local hrp = character and character:FindFirstChild("HumanoidRootPart")
	if hrp then
		local pos = hrp.Position
		coordLabel.Text = string.format("พิกัดปัจจุบัน:\nX: %d  Y: %d  Z: %d", pos.X, pos.Y, pos.Z)
	end
end)
